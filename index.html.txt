    <script>
        function triggerPopup(text, event) {
            const popup = document.createElement('div');
            popup.className = 'floating-popup';
            popup.innerText = text;
            popup.style.left = event.clientX + 'px';
            popup.style.top = event.clientY + 'px';
            document.body.appendChild(popup);
            setTimeout(() => { popup.remove(); }, 1200);
        }

        function nextQuestion(currentQ) {
            setTimeout(() => {
                document.getElementById('q' + currentQ).classList.remove('active');
                if (currentQ < 6) {
                    document.getElementById('q' + (currentQ + 1)).classList.add('active');
                } else {
                    document.getElementById('part1').style.display = 'none';
                    document.getElementById('part2').style.display = 'block';
                    document.getElementById('finishBtn').style.display = 'block';
                }
            }, 500); 
        }

        function goBack(targetQ) {
            const blocks = document.querySelectorAll('.question-block');
            for(let i = 0; i < blocks.length; i++) {
                blocks[i].classList.remove('active');
            }
            document.getElementById('q' + targetQ).classList.add('active');
        }

        function handleQ1(event, answer) {
            triggerPopup('LOL', event);
            if(answer === 'no') {
                nextQuestion(1);
            }
        }

        function handleQ2(event, isCorrect) {
            if(isCorrect) {
                triggerPopup('What are you doing, FOCUS 🤨', event);
                nextQuestion(2);
            } else {
                triggerPopup('HEHEHE', event);
            }
        }

        function handleGeneric(event, qNum, isCorrect, customSuccessText) {
            if(isCorrect) {
                let successMessage = customSuccessText ? customSuccessText : '😘';
                triggerPopup(successMessage, event);
                nextQuestion(qNum);
            } else {
                triggerPopup('😾', event);
            }
        }

        function selectOption(btn) {
            const siblings = btn.parentElement.children;
            for(let i=0; i<siblings.length; i++) {
                siblings[i].classList.remove('selected');
            }
            btn.classList.add('selected');
        }

        function finishSurvey() {
            let resultsText = "Hey! Here are my choices for our date! ❤️\n\n";
            
            const items = document.querySelectorAll('.survey-item');
            items.forEach(item => {
                const questionTitle = item.querySelector('h3').innerText;
                const selectedBtn = item.querySelector('.selected');
                const answer = selectedBtn ? selectedBtn.innerText : "I couldn't decide!";
                resultsText += `*${questionTitle}*: ${answer}\n`;
            });

            const encodedText = encodeURIComponent(resultsText); 

            document.getElementById('part2').innerHTML = `
                <h2 class="part2-header">Date officially secured! 🎉</h2>
                <p>I am so excited! Click the button below to open WhatsApp and send your choices to me.</p>
                <a href="https://wa.me/?text=${encodedText}" 
                   style="display: inline-block; padding: 15px 30px; background-color: #25D366; color: white; text-decoration: none; border-radius: 10px; font-weight: bold; font-size: 1.1rem; margin-top: 20px;">
                   Send Answers via WhatsApp 📱
                </a>
            `;
        }
